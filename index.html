/* ================================== 1/100秒カウントダウン ================================== */ function startCountdown() { // 10秒スタート const duration = 10 * 1000; // 開始時刻 const startTime = performance.now(); const display = $("countdown"); function updateCountdown(currentTime) { // 経過時間 const elapsed = currentTime - startTime; // 残り時間 const remaining = Math.max( 0, duration - elapsed ); // 秒に変換して小数点以下2桁 const seconds = (remaining / 1000).toFixed(2); display.textContent = seconds; // 0.00になったら終了 if (remaining <= 0) { // 最後の振動 vibrate([ 200, 100, 200, 100, 500 ]); // 赤フラッシュ document.body.classList.add("flash-red"); setTimeout(() => { document.body.classList.remove( "flash-red" ); show("screen3"); }, 700); // INTRUDER画面 setTimeout(() => { show("screen4"); vibrate(100); }, 2600); // 最終オチ setTimeout(() => { show("screen5"); }, 6500); return; } // 次のフレームで更新 requestAnimationFrame(updateCountdown); } requestAnimationFrame(updateCountdown); }
